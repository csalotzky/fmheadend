cmake_minimum_required(VERSION 3.16)
project(fmheadend_muxd VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(mux_core src/core.cpp)
target_include_directories(mux_core PUBLIC include)
target_link_libraries(mux_core PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mux_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(fmheadend-muxd src/main.cpp src/server.cpp src/transport.cpp)
  target_link_libraries(fmheadend-muxd PRIVATE mux_core)
  target_compile_options(fmheadend-muxd PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS fmheadend-muxd RUNTIME DESTINATION bin)
else()
  message(STATUS "Linux sockets/spidev daemon excluded on this platform; core tests remain available")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_mux_transport tests/test_transport.cpp src/transport.cpp)
  target_link_libraries(test_mux_transport PRIVATE mux_core)
  add_test(NAME transport_prefix COMMAND test_mux_transport)
  add_executable(test_mux_core tests/test_core.cpp)
  target_link_libraries(test_mux_core PRIVATE mux_core)
  foreach(case ring underrun overrun fragmented frame counter disconnect capacity)
    add_test(NAME ${case} COMMAND test_mux_core ${case})
  endforeach()
  if(TARGET fmheadend-muxd)
    add_executable(test_mux_control tests/test_control.cpp src/server.cpp)
    target_link_libraries(test_mux_control PRIVATE mux_core)
    target_compile_options(test_mux_control PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME control_boundaries COMMAND test_mux_control)
    set_tests_properties(control_boundaries PROPERTIES TIMEOUT 10)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME unix_socket_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sockets.py $<TARGET_FILE:fmheadend-muxd>)
    set_tests_properties(unix_socket_integration PROPERTIES TIMEOUT 30)
    add_test(NAME buffer_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:fmheadend-muxd>)
    set_tests_properties(buffer_cli PROPERTIES TIMEOUT 15)
  endif()
endif()
