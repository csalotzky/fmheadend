cmake_minimum_required(VERSION 3.13)
project(fmheadend_firmware_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME test_i2s_timing COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_i2s_timing.py)
endif()

add_executable(test_qn_bus test_qn_bus.c ../qn_bus.c)
add_executable(test_safe_boot test_safe_boot.c ../qn_boot.c)
add_executable(test_host_protocol test_host_protocol.c ../host_protocol.c ../qn_control.c fake_qn.c)
add_executable(test_host_i2c test_host_i2c.c ../host_protocol.c ../qn_control.c fake_qn.c)
add_executable(test_qn_control test_qn_control.c ../qn_control.c ../host_protocol.c fake_qn.c)
add_executable(test_audio_samples test_audio_samples.c ../audio_samples.c)
add_executable(test_audio_engine test_audio_engine.c ../audio_samples.c)
target_include_directories(test_audio_engine BEFORE PRIVATE audio_fakes)
if(UNIX)
    target_link_libraries(test_audio_samples m)
endif()
target_compile_definitions(test_host_i2c PRIVATE FMHEADEND_HOST_I2C_ADDRESS=0x62 TEST_ADAPTER_CLOCK=1)
foreach(target test_qn_bus test_safe_boot test_host_protocol test_host_i2c test_qn_control test_audio_samples test_audio_engine)
    target_include_directories(${target} PRIVATE . ..)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    elseif(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 10)
endforeach()
