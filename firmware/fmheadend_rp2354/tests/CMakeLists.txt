# Population endpoints/intermediate variants, including non-power-of-two packets.
foreach(spec 1_128 4_128 8_128 1_64 4_64 8_64 4_48 1_1)
    string(REPLACE "_" ";" dimensions ${spec})
    list(GET dimensions 0 tx)
    list(GET dimensions 1 frames)
    set(t test_variant_${spec})
    add_executable(${t} test_variant.c ../spi_stream.c ../audio_samples.c ../host_protocol.c ../qn_control.c ../rds_fifo.c fake_qn.c)
    target_include_directories(${t} PRIVATE . ..)
    target_compile_definitions(${t} PRIVATE TX_COUNT=${tx} FRAMES_PER_PACKET=${frames})
    target_compile_options(${t} PRIVATE -UNDEBUG)
    add_test(NAME ${t} COMMAND ${t})
endforeach()
